// ---------- Question 14 ----------
// var a = 2, b = 1;
// var result = --a - --b + ++b + b--;
//
// Explanation of each stage:
// --a;                    -> a becomes 1, value used = 1
// --a - --b;              -> b becomes 0, so 1 - 0 = 1
// --a - --b + ++b;        -> b becomes 1, so 1 + 1 = 2
// --a - --b + ++b + b--;  -> b-- uses 1 then b becomes 0, so 2 + 1 = 3
//
// Final: a = 1, b = 0, result = 3

var x = 2, y = 1;
var result = --x - --y + ++y + y--;
document.write("a is " + x + "<br>");
document.write("b is " + y + "<br>");
document.write("result is " + result + "<br><br>");
