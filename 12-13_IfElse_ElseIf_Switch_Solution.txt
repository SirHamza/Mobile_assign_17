// ---------- Question 6 ----------
// Number, uppercase or lowercase (using ASCII codes)
var ch = prompt("Enter a single character");
var code = ch.charCodeAt(0);

if (code >= 48 && code <= 57) {
    alert(ch + " is a number");
} else if (code >= 65 && code <= 90) {
    alert(ch + " is an uppercase letter");
} else if (code >= 97 && code <= 122) {
    alert(ch + " is a lowercase letter");
} else {
    alert(ch + " is a special character");
}


// ---------- Question 7 ----------
// Calculator using switch
var number1 = parseFloat(prompt("Enter first number"));
var number2 = parseFloat(prompt("Enter second number"));
var operator = prompt("Enter operator (+, -, *, /, %)");

switch (operator) {
    case "+":
        alert("Result: " + (number1 + number2));
        break;
    case "-":
        alert("Result: " + (number1 - number2));
        break;
    case "*":
        alert("Result: " + (number1 * number2));
        break;
    case "/":
        alert("Result: " + (number1 / number2));
        break;
    case "%":
        alert("Result: " + (number1 % number2));
        break;
    default:
        alert("Invalid operator");
}
